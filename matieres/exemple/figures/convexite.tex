% Corde au-dessus du graphe d'une fonction convexe.
\begin{tikzpicture}[scale=1.3, >=Stealth]
  \draw[->] (-0.3,0) -- (4.2,0) node[right] {$x$};
  \draw[->] (0,-0.3) -- (0,3.4);
  \draw[thick, blue, domain=0.2:3.9, samples=60] plot (\x, {0.25*(\x-1.6)^2+0.5});
  \coordinate (A) at (0.6, {0.25*(0.6-1.6)^2+0.5});
  \coordinate (B) at (3.6, {0.25*(3.6-1.6)^2+0.5});
  \draw[thick, red] (A) -- (B);
  \fill (A) circle (1.5pt) node[above left] {$f(x)$};
  \fill (B) circle (1.5pt) node[above right] {$f(y)$};
  \draw[dashed] (A) -- (0.6,0) node[below] {$x$};
  \draw[dashed] (B) -- (3.6,0) node[below] {$y$};
  \node[red, align=center] at (2.1,2.4) {$\lambda f(x) + (1-\lambda) f(y)$};
\end{tikzpicture}
